import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import warnings
import re
import sys

from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.ensemble import IsolationForest
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import confusion_matrix
from mlxtend.frequent_patterns import apriori, association_rules

# Ignorar advertencias visuales de las librerías
warnings.filterwarnings('ignore')

# ---------------------------------------------------------
# 1. INTRODUCCIÓN AL ANÁLISIS
# ---------------------------------------------------------
print("="*80)
print("👔 ANÁLISIS DE MINERÍA DE DATOS - BASE DE DATOS ZARA")
print("="*80)
print("INTRODUCCIÓN:")
print("Este sistema realiza un análisis integral del catálogo de Zara empleando las 4 principales")
print("técnicas de Minería de Datos: Detección de Outliers, Clustering, Reglas de Asociación y Clasificación.")
print("El objetivo es descubrir patrones ocultos de ventas, identificar los perfiles de productos rentables,")
print("y predecir el éxito de ventas basado en características como precio, temporada y promociones.\n")

# ---------------------------------------------------------
# 2. CARGA DE DATOS (Soporte Inteligente para Colab o Local/GitHub)
# ---------------------------------------------------------
try:
    from google.colab import files
    print("Entorno Google Colab detectado. 📥 Sube tu archivo (zara.xlsx o zara.csv):")
    uploaded = files.upload()
    file_name = list(uploaded.keys())[0]
except ImportError:
    print("Entorno local / GitHub detectado.")
    file_name = input("📥 Ingresa la ruta exacta de tu archivo (ej. zara.csv): ").strip()
    file_name = file_name.strip('\"\'') # Limpiar comillas si arrastran el archivo a la terminal

try:
    if file_name.endswith('.csv'):
        try:
            df = pd.read_csv(file_name, sep=';', on_bad_lines='skip')
            if len(df.columns) <= 1: df = pd.read_csv(file_name, sep=',', on_bad_lines='skip')
        except:
            df = pd.read_csv(file_name, sep=',', on_bad_lines='skip')
    else:
        df = pd.read_excel(file_name)
    print("✅ ¡Archivo cargado y leído exitosamente!\n")
except Exception as e:
    print(f"❌ Error crítico al cargar el archivo. Verifica la ruta o formato: {e}")
    sys.exit()

# ---------------------------------------------------------
# 3. LIMPIEZA EXTREMA Y PREPARACIÓN DE VARIABLES
# ---------------------------------------------------------
col_map = {str(c).strip().lower(): c for c in df.columns}
price_col = col_map.get('price', col_map.get('precio'))
sales_col = col_map.get('sales volume', col_map.get('ventas'))
promo_col = col_map.get('promotion', 'Promotion')
season_col = col_map.get('seasonal', 'Seasonal')
sec_col = col_map.get('section', 'section')

if not price_col or not sales_col:
    print("❌ Faltan columnas clave (Precio o Ventas) en el archivo.")
    sys.exit()

# Función para extraer números reales evitando ceros falsos o errores de coma
def clean_currency(x):
    if pd.isna(x): return np.nan
    x_str = str(x).strip().replace('$', '').replace('€', '').replace(' ', '')
    if ',' in x_str and '.' in x_str:
        x_str = x_str.replace('.', '').replace(',', '.') if x_str.rfind(',') > x_str.rfind('.') else x_str.replace(',', '')
    elif ',' in x_str:
        x_str = x_str.replace(',', '.')
    match = re.search(r'\d+\.?\d*', x_str)
    return float(match.group()) if match else np.nan

df[price_col] = df[price_col].apply(clean_currency)
df[sales_col] = df[sales_col].apply(clean_currency)
df = df.dropna(subset=[price_col, sales_col]).copy()
df = df[df[price_col] > 0] # Filtro de seguridad (sin $0)

df['Ingresos_Brutos'] = df[price_col] * df[sales_col]

# Discretizar (Categorizar) para algoritmos categóricos
med_price = df[price_col].median()
med_sales = df[sales_col].median()
df['Nivel_Precio'] = np.where(df[price_col] > med_price, 'Precio Alto', 'Precio Bajo')
df['Nivel_Ventas'] = np.where(df[sales_col] > med_sales, 'Ventas Altas', 'Ventas Bajas')

sns.set_theme(style="whitegrid", palette="muted")
print("⏳ Datos limpios, normalizados y procesados. Iniciando Minería de Datos...\n")

# =========================================================
# TÉCNICA 1: DETECCIÓN DE OUTLIERS (ANOMALÍAS)
# =========================================================
print("🔍 1. DETECCIÓN DE OUTLIERS (ISOLATION FOREST)...")
iso_forest = IsolationForest(contamination=0.05, random_state=42)
X_out = df[[price_col, sales_col]]
df['Outlier'] = iso_forest.fit_predict(X_out)
df['Outlier_Label'] = df['Outlier'].map({1: 'Normal', -1: 'Anomalía (Outlier)'})

# GRÁFICA 1: Scatter Outliers
plt.figure(figsize=(10, 6))
sns.scatterplot(data=df, x=price_col, y=sales_col, hue='Outlier_Label', 
                palette={'Normal': '#3498db', 'Anomalía (Outlier)': '#e74c3c'}, 
                size='Outlier_Label', sizes={'Normal': 40, 'Anomalía (Outlier)': 120}, alpha=0.8, edgecolor='black')
plt.title('Gráfica 1 (Outliers): Anomalías detectadas con Isolation Forest', fontsize=14, fontweight='bold')
plt.xlabel('Precio ($)')
plt.ylabel('Volumen de Ventas')
plt.show()

# GRÁFICA 2: Boxplot Outliers
plt.figure(figsize=(10, 6))
sns.boxplot(data=df, x=sec_col, y=price_col, palette='Pastel2')
sns.stripplot(data=df[df['Outlier'] == -1], x=sec_col, y=price_col, color="red", size=8, jitter=True, label="Outliers", edgecolor='black', linewidth=1)
plt.title('Gráfica 2 (Outliers): Boxplot de Precios resaltando Anomalías Premium', fontsize=14, fontweight='bold')
plt.xlabel('Sección del Catálogo')
plt.ylabel('Precio ($)')
plt.show()

# =========================================================
# TÉCNICA 2: CLUSTERING (K-MEANS)
# =========================================================
print("🧩 2. CLUSTERING (K-MEANS)...")
X_scaled = StandardScaler().fit_transform(X_out)
kmeans = KMeans(n_clusters=4, random_state=42, n_init=10)
df['Cluster'] = kmeans.fit_predict(X_scaled)
df['Cluster_Label'] = 'Clúster ' + df['Cluster'].astype(str)

# GRÁFICA 3: K-Means 2D
plt.figure(figsize=(10, 6))
sns.scatterplot(data=df, x=price_col, y=sales_col, hue='Cluster_Label', palette='Set1', size=sales_col, sizes=(40, 200), alpha=0.8, edgecolor='black')
plt.title('Gráfica 3 (Clustering): Segmentación Estratégica 2D', fontsize=14, fontweight='bold')
plt.xlabel('Precio ($)')
plt.ylabel('Volumen de Ventas')
plt.show()

# GRÁFICA 4: K-Means 3D
fig = plt.figure(figsize=(12, 8))
ax = fig.add_subplot(111, projection='3d')
sc = ax.scatter(df[price_col], df[sales_col], df['Ingresos_Brutos'], c=df['Cluster'], cmap='Set1', s=60, alpha=0.9, edgecolor='black')
ax.set_title('Gráfica 4 (Clustering): Segmentación 3D (Precio vs Ventas vs Ingresos)', fontsize=14, fontweight='bold')
ax.set_xlabel('Precio ($)')
ax.set_ylabel('Ventas (Unidades)')
ax.set_zlabel('Ingresos Brutos ($)')
ax.view_init(elev=20, azim=130)
plt.show()

# =========================================================
# TÉCNICA 3: REGLAS DE ASOCIACIÓN (APRIORI)
# =========================================================
print("🔗 3. REGLAS DE ASOCIACIÓN (ALGORITMO APRIORI)...")
# Preparar "Cesta" para el algoritmo
basket = df[['Nivel_Precio', 'Nivel_Ventas', season_col, promo_col, sec_col]].astype(str)
basket_encoded = pd.get_dummies(basket)
freq_items = apriori(basket_encoded, min_support=0.05, use_colnames=True)

if not freq_items.empty:
    rules = association_rules(freq_items, metric="confidence", min_threshold=0.6)
    rules = rules.sort_values(by=['lift'], ascending=False)
    print("\n--- Top 3 Reglas de Asociación Matemáticas Descubiertas ---")
    for idx, row in rules.head(3).iterrows():
        ant = ", ".join(list(row['antecedents'])).replace(f'{season_col}_','').replace(f'{promo_col}_','').replace(f'{sec_col}_','')
        con = ", ".join(list(row['consequents'])).replace(f'{season_col}_','').replace(f'{promo_col}_','').replace(f'{sec_col}_','')
        print(f"SI [{ant}] => ENTONCES [{con}] (Confianza: {row['confidence']*100:.1f}%)")
print("\n")

# GRÁFICA 5: Asociación Promociones
plt.figure(figsize=(10, 6))
sns.barplot(data=df, x=sec_col, y=sales_col, hue=promo_col, errorbar=None, edgecolor='black', palette='viridis')
plt.title('Gráfica 5 (Asociación): Regla de Promoción Hombres vs Mujeres', fontsize=14, fontweight='bold')
plt.xlabel('Sección')
plt.ylabel('Ventas Promedio')
plt.show()

# GRÁFICA 6: Asociación Temporadas
plt.figure(figsize=(10, 6))
sns.barplot(data=df, x=season_col, y=sales_col, hue=promo_col, errorbar=None, edgecolor='black', palette='magma')
plt.title('Gráfica 6 (Asociación): Regla del impacto Promo en Ropa Seasonal', fontsize=14, fontweight='bold')
plt.xlabel('¿Es ropa de Temporada?')
plt.ylabel('Ventas Promedio')
plt.show()

# GRÁFICA 7: Heatmap de Zonas Calientes
plt.figure(figsize=(8, 5))
pivot = df.pivot_table(values=sales_col, index='Nivel_Precio', columns=sec_col, aggfunc='mean').fillna(0)
sns.heatmap(pivot, annot=True, fmt=".0f", cmap="YlOrRd", linewidths=.5, cbar_kws={'label': 'Volumen de Ventas'})
plt.title('Gráfica 7 (Asociación): Zonas Calientes (Precio vs Sección)', fontsize=14, fontweight='bold')
plt.show()

# =========================================================
# TÉCNICA 4: CLASIFICACIÓN (ÁRBOL DE DECISIÓN)
# =========================================================
print("🌳 4. CLASIFICACIÓN (MODELO PREDICTIVO: ÁRBOL DE DECISIÓN)...")
# El objetivo es predecir matemáticamente si una prenda tendrá "Ventas Altas" (1) o "Ventas Bajas" (0)
df['Target_Ventas'] = np.where(df[sales_col] > med_sales, 1, 0)
df_clf = df.dropna(subset=[promo_col, season_col, sec_col]).copy()

# Transformar categóricas a numéricas para entrenar la IA
le_promo, le_season, le_sec = LabelEncoder(), LabelEncoder(), LabelEncoder()
X_class = pd.DataFrame()
X_class['Precio'] = df_clf[price_col]
X_class['Promocion'] = le_promo.fit_transform(df_clf[promo_col].astype(str))
X_class['Es_Temporada'] = le_season.fit_transform(df_clf[season_col].astype(str))
X_class['Seccion'] = le_sec.fit_transform(df_clf[sec_col].astype(str))
y_class = df_clf['Target_Ventas']

# Separar datos de entrenamiento y prueba
X_train, X_test, y_train, y_test = train_test_split(X_class, y_class, test_size=0.3, random_state=42)

# Entrenar Árbol de Decisión
clf = DecisionTreeClassifier(max_depth=4, random_state=42)
clf.fit(X_train, y_train)

# GRÁFICA 8: Feature Importances
plt.figure(figsize=(9, 5))
sns.barplot(x=clf.feature_importances_, y=X_class.columns, palette='Blues_r', edgecolor='black')
plt.title('Gráfica 8 (Clasificación): ¿Qué variables predicen el éxito de ventas?', fontsize=14, fontweight='bold')
plt.xlabel('Nivel de Importancia Predictiva')
plt.show()

# GRÁFICA 9: Matriz de Confusión
y_pred = clf.predict(X_test)
cm = confusion_matrix(y_test, y_pred)
plt.figure(figsize=(6, 5))
sns.heatmap(cm, annot=True, fmt='d', cmap='Greens', cbar=False, xticklabels=['Ventas Bajas', 'Ventas Altas'], yticklabels=['Bajas (Real)', 'Altas (Real)'])
plt.title('Gráfica 9 (Clasificación): Matriz de Confusión del Modelo', fontsize=14, fontweight='bold')
plt.show()

# GRÁFICA 10: Violinplot General de Soporte
plt.figure(figsize=(12, 6))
sns.violinplot(data=df, x=promo_col, y=sales_col, hue=sec_col, inner="quartile", palette="Set3")
plt.title('Gráfica 10 (Distribución General): Densidad de Ventas y Secciones', fontsize=14, fontweight='bold')
plt.xlabel('¿Está en Promoción?')
plt.ylabel('Volumen de Ventas (Unidades)')
plt.show()
